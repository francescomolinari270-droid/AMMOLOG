      let state = loadState();

      function loadState() {
        try {
          const raw = localStorage.getItem(STORAGE_KEY);
          if (!raw) {
            const initial = getDefaultState();
            localStorage.setItem(STORAGE_KEY, JSON.stringify(initial));
            return initial;
          }
          const parsed = JSON.parse(raw);
          if (!Array.isArray(parsed.weapons) || !Array.isArray(parsed.lots)) {
            throw new Error("Struttura non valida");
          }
          return {
            weapons: parsed.weapons,
            lots: parsed.lots,
            logs: Array.isArray(parsed.logs) ? parsed.logs : []
          };
        } catch (err) {
          const fallback = getDefaultState();
          localStorage.setItem(STORAGE_KEY, JSON.stringify(fallback));
          return fallback;
        }
      }

      function saveState() {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      }

      function showToast(msg) {
        const toast = document.getElementById("toast");
        toast.textContent = msg;
        toast.classList.add("show");
        clearTimeout(showToast._timer);
        showToast._timer = setTimeout(() => {
          toast.classList.remove("show");
        }, 2800);
      }

      function addLog(title, detail, delta) {
        state.logs.unshift({
          id: "log-" + Date.now() + "-" + Math.random().toString(36).slice(2, 6),
          timestamp: new Date().toISOString(),
          title,
          detail,
          delta
        });
        if (state.logs.length > 200) {
          state.logs.length = 200;
        }
      }

      function getCategoryTotals() {
        let shortTotal = 0;
        let huntTotal = 0;
        for (const lot of state.lots) {
          const qty = Number(lot.quantity) || 0;
          if (getCategoryForCaliber(lot.caliber) === "short") {
            shortTotal += qty;
          } else {
            huntTotal += qty;
          }
        }
        return { shortTotal, huntTotal };
      }

      function getRemainingCapacity(caliber) {
        const { shortTotal, huntTotal } = getCategoryTotals();
        if (getCategoryForCaliber(caliber) === "short") {
          return Math.max(0, LIMIT_SHORT - shortTotal);
        }
        return Math.max(0, LIMIT_HUNT - huntTotal);
      }

      function renderLogs() {
        const list = document.getElementById("logsList");
        if (!state.logs.length) {
          list.innerHTML = `<div class="empty-state">Nessun movimento registrato.</div>`;
          return;
        }

        list.innerHTML = state.logs.map((entry) => {
          const dt = new Date(entry.timestamp);
          const formattedDate = isNaN(dt.getTime())
            ? ""
            : dt.toLocaleDateString("it-IT", { day: "2-digit", month: "2-digit", year: "numeric" }) +
              " " +
              dt.toLocaleTimeString("it-IT", { hour: "2-digit", minute: "2-digit" });

          const deltaNum = Number(entry.delta) || 0;
          let deltaHtml = "";
          if (deltaNum < 0) {
            deltaHtml = `<div class="log-delta delta-minus">${deltaNum.toLocaleString("it-IT")} colpi</div>`;
          } else if (deltaNum > 0) {
            deltaHtml = `<div class="log-delta delta-plus">+${deltaNum.toLocaleString("it-IT")} colpi</div>`;
          }

          return `
            <div class="log-item">
              <div class="log-main">
                <div class="log-title">${escapeHtml(entry.title)}</div>
                <div class="log-sub">${escapeHtml(entry.detail)}${formattedDate ? " • " + formattedDate : ""}</div>
              </div>
              ${deltaHtml}
            </div>
          `;
        }).join("");
      }

      function escapeHtml(str) {
        return String(str || "")
          .replace(/&/g, "&")
          .replace(/</g, "<")
          .replace(/>/g, ">")
          .replace(/"/g, """);
      }

      function populateShootModal(preselectedWeaponId) {
        const weaponSelect = document.getElementById("shootWeapon");
        if (!state.weapons.length) {
          showToast("Aggiungi prima un'arma all'arsenale.");
          return;
        }

        weaponSelect.innerHTML = state.weapons
          .map((w) => `<option value="${w.id}">${escapeHtml(w.name)} (${escapeHtml(w.caliber)})</option>`)
          .join("");

        if (preselectedWeaponId && state.weapons.some((w) => w.id === preselectedWeaponId)) {
          weaponSelect.value = preselectedWeaponId;
        }

        updateShootLotsForSelectedWeapon();
        openModal("shootModal");
      }

      function updateShootLotsForSelectedWeapon() {
        const weaponId = document.getElementById("shootWeapon").value;
        const weapon = state.weapons.find((w) => w.id === weaponId);
        const lotSelect = document.getElementById("shootLot");

        if (!weapon) {
          lotSelect.innerHTML = `<option value="">Nessuna arma</option>`;
          return;
        }

        const compatibleLots = state.lots.filter((l) => l.caliber === weapon.caliber && Number(l.quantity) > 0);
        if (!compatibleLots.length) {
          lotSelect.innerHTML = `<option value="">Nessun lotto disponibile per ${escapeHtml(weapon.caliber)}</option>`;
          return;
        }

        lotSelect.innerHTML = compatibleLots
          .map((l) => `<option value="${l.id}">${escapeHtml(l.brand)} — ${Number(l.quantity)} colpi disp.</option>`)
          .join("");
      }

      document.getElementById("shootForm").addEventListener("submit", (e) => {
        e.preventDefault();
        const weaponId = document.getElementById("shootWeapon").value;
        const lotId = document.getElementById("shootLot").value;
        const qty = parseInt(document.getElementById("shootQty").value, 10);
        const note = document.getElementById("shootNote").value.trim();

        const weapon = state.weapons.find((w) => w.id === weaponId);
        const lot = state.lots.find((l) => l.id === lotId);

        if (!weapon || !lot) {
          showToast("Seleziona un'arma e un lotto valido.");
          return;
        }
        if (!Number.isFinite(qty) || qty <= 0) {
          showToast("Inserisci un numero di colpi valido.");
          return;
        }
        if (qty > lot.quantity) {
          showToast(`Colpi insufficienti nel lotto (disponibili: ${lot.quantity}).`);
          return;
        }

        lot.quantity -= qty;
        weapon.totalFired = (Number(weapon.totalFired) || 0) + qty;
        weapon.shotsSinceClean = (Number(weapon.shotsSinceClean) || 0) + qty;

        addLog(
          `Sessione con ${weapon.name}`,
          `Lotto ${lot.caliber} ${lot.brand}${note ? " — " + note : ""}`,
          -qty
        );

        saveState();
        renderAll();
        closeModal("shootModal");
        showToast(`Scaricati ${qty} colpi (${weapon.name}).`);
      });

          state.weapons = state.weapons.filter((w) => w.id !== id);
          saveState();
          renderWeapons();

      document.getElementById("btnExportJson").addEventListener("click", () => {
        const blob = new Blob([JSON.stringify(state, null, 2)], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `ammolog_backup_${new Date().toISOString().slice(0, 10)}.json`;
        a.click();
        URL.revokeObjectURL(url);
        showToast("Backup JSON esportato.");
      });

      const importInput = document.getElementById("importFileInput");
      document.getElementById("btnImportJson").addEventListener("click", () => importInput.click());

      importInput.addEventListener("change", (e) => {
        const file = e.target.files && e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const data = JSON.parse(reader.result);
            if (!Array.isArray(data.weapons) || !Array.isArray(data.lots)) {
              throw new Error("Formato non valido");
            }
            state = {
              weapons: data.weapons,
              lots: data.lots,
              logs: Array.isArray(data.logs) ? data.logs : []
            };
            saveState();
            renderAll();
            showToast("Backup ripristinato.");
          } catch (err) {
            showToast("File JSON non valido.");
          }
          importInput.value = "";
        };
        reader.readAsText(file);
      });

      document.getElementById("btnResetDemo").addEventListener("click", () => {
        state = getDefaultState();
        saveState();
        renderAll();
        showToast("Dati base ripristinati.");
      });

      // Setup PWA (Manifest dinamico + Service Worker su protocolli http/https)
      function setupPwa() {
        try {
          const manifestData = {
            name: "AMMOLOG",
            short_name: "AMMOLOG",
            start_url: ".",
            display: "standalone",
            background_color: "#0f172a",
            theme_color: "#0f172a"
          };
          const manifestBlob = new Blob([JSON.stringify(manifestData)], { type: "application/manifest+json" });
          const manifestUrl = URL.createObjectURL(manifestBlob);
          const link = document.createElement("link");
          link.rel = "manifest";
          link.href = manifestUrl;
          document.head.appendChild(link);
        } catch (_) {}

        if ("serviceWorker" in navigator && (window.location.protocol === "https:" || window.location.hostname === "localhost")) {
          const swCode = `
            const CACHE_NAME = "ammolog-cache-v1";
            self.addEventListener("install", (event) => {
              event.waitUntil(
                caches.open(CACHE_NAME).then((cache) => cache.addAll([self.registration.scope]))
              );
              self.skipWaiting();
            });
            self.addEventListener("fetch", (event) => {
              if (event.request.method !== "GET") return;
              event.respondWith(
                caches.match(event.request).then((cached) => cached || fetch(event.request))
              );
            });
          `;
          const swBlob = new Blob([swCode], { type: "application/javascript" });
          const swUrl = URL.createObjectURL(swBlob);
          navigator.serviceWorker.register(swUrl).catch(() => {});
        }
      }

      setupPwa();
      renderAll();
    })();
  </script>
